# 第 10 章：管理后台

本章制作真正可用的后台，不引入第二套前端工程。公共页和后台共享 `web`、OpenAPI 类型、Design Tokens 与构建产物；后台只在浏览器运行，页面仍能被 Next.js 静态导出。Access Token 只保存在 JavaScript 内存，刷新凭证只存在 HttpOnly Cookie，刷新页面后由 `/auth/refresh` 恢复会话。

## 10-03 创建内存会话仓库

创建 `web/lib/admin/session.ts`，文件全文：

```typescript
'use client';

import { useSyncExternalStore } from 'react';
import type { User } from '@/lib/admin/types';

export type SessionStatus = 'booting' | 'authenticated' | 'anonymous';
type SessionState = { status: SessionStatus; accessToken: string | null; user: User | null };

let state: SessionState = { status: 'booting', accessToken: null, user: null };
const listeners = new Set<() => void>();

function emit(next: SessionState): void {
  state = next;
  listeners.forEach((listener) => listener());
}

function subscribe(listener: () => void): () => void {
  listeners.add(listener);
  return () => listeners.delete(listener);
}

export const session = {
  get: () => state,
  authenticate: (accessToken: string, user: User | null = state.user) =>
    emit({ status: 'authenticated', accessToken, user }),
  setUser: (user: User) => emit({ ...state, status: 'authenticated', user }),
  clear: () => emit({ status: 'anonymous', accessToken: null, user: null }),
};

const serverSnapshot: SessionState = { status: 'booting', accessToken: null, user: null };

export function useSession(): SessionState {
  return useSyncExternalStore(subscribe, session.get, () => serverSnapshot);
}
```

不要增加 `localStorage` 或 `sessionStorage`。XSS 一旦发生，长期持久化 Token 会扩大影响；刷新 Cookie 的 HttpOnly 属性使 JavaScript 无法读取它。

```typescript
'use client';

import { session } from '@/lib/admin/session';
import type {
  AccessToken,
  ApiErrorBody,
  ArticleCreateRequest,
  ArticleDetail,
  ArticleListResponse,
  ArticleUpdateRequest,
  Category,
  CategoryRequest,
  DashboardStats,
  Envelope,
  ProjectCreateRequest,
  ProjectDetail,
  ProjectListResponse,
  ProjectUpdateRequest,
  SiteSettings,
  Tag,
  TagRequest,
  User,
} from '@/lib/admin/types';

const baseURL = process.env.NEXT_PUBLIC_API_BASE_URL ?? 'http://localhost:8080/api/v1';
let refreshPromise: Promise<string | null> | null = null;

export class ApiError extends Error {
  constructor(
    public readonly status: number,
    public readonly code: number,
    message: string,
  ) {
    super(message);
    this.name = 'ApiError';
  }
}

async function parse<T>(response: Response): Promise<T> {
  const body = (await response.json().catch(() => null)) as Envelope<T> | ApiErrorBody | null;
  if (!response.ok || body === null || body.code !== 0) {
    const error = body as ApiErrorBody | null;
    throw new ApiError(
      response.status,
      error?.code ?? response.status,
      error?.message ?? '请求失败',
    );
  }
  return (body as Envelope<T>).data;
}

async function refresh(): Promise<string | null> {
  if (refreshPromise !== null) return refreshPromise;
  refreshPromise = fetch(`${baseURL}/auth/refresh`, {
    method: 'POST',
    credentials: 'include',
    headers: { Accept: 'application/json', 'X-Requested-With': 'XMLHttpRequest' },
  })
    .then(async (response) => {
      if (!response.ok) return null;
      const token = await parse<AccessToken>(response);
      session.authenticate(token.access_token);
      return token.access_token;
    })
    .catch(() => null)
    .finally(() => {
      refreshPromise = null;
    });
  return refreshPromise;
}

type RequestOptions = Omit<RequestInit, 'body'> & {
  body?: unknown;
  auth?: boolean;
  retry?: boolean;
};

async function request<T>(path: string, options: RequestOptions = {}): Promise<T> {
  const { body, auth = false, retry = true, headers, ...init } = options;
  const token = session.get().accessToken;
  const response = await fetch(`${baseURL}${path}`, {
    ...init,
    credentials: 'include',
    headers: {
      Accept: 'application/json',
      ...(body === undefined ? {} : { 'Content-Type': 'application/json' }),
      ...(init.method && init.method !== 'GET' ? { 'X-Requested-With': 'XMLHttpRequest' } : {}),
      ...(auth && token ? { Authorization: `Bearer ${token}` } : {}),
      ...headers,
    },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  if (response.status === 401 && auth && retry) {
    const nextToken = await refresh();
    if (nextToken !== null) return request<T>(path, { ...options, retry: false });
    session.clear();
  }
  return parse<T>(response);
}

export const adminApi = {
  login: (username: string, password: string) =>
    request<AccessToken>('/auth/login', { method: 'POST', body: { username, password } }),
  restore: refresh,
  me: () => request<User>('/auth/me', { auth: true }),
  logout: () =>
    request<unknown>('/auth/logout', { method: 'POST', auth: true }).finally(session.clear),
  dashboard: () => request<DashboardStats>('/admin/dashboard', { auth: true }),

  articles: () =>
    request<ArticleListResponse>('/admin/articles?page=1&page_size=100', { auth: true }),
  article: (slug: string) =>
    request<ArticleDetail>(`/articles/${encodeURIComponent(slug)}`, { auth: true }),
  createArticle: (body: ArticleCreateRequest) =>
    request<ArticleDetail>('/articles', { method: 'POST', auth: true, body }),
  updateArticle: (slug: string, body: ArticleUpdateRequest) =>
    request<ArticleDetail>(`/articles/${encodeURIComponent(slug)}`, {
      method: 'PUT',
      auth: true,
      body,
    }),
  deleteArticle: (slug: string) =>
    request<unknown>(`/articles/${encodeURIComponent(slug)}`, { method: 'DELETE', auth: true }),
  publishArticle: (slug: string) =>
    request<ArticleDetail>(`/articles/${encodeURIComponent(slug)}/publish`, {
      method: 'POST',
      auth: true,
    }),
  unpublishArticle: (slug: string) =>
    request<ArticleDetail>(`/articles/${encodeURIComponent(slug)}/unpublish`, {
      method: 'POST',
      auth: true,
    }),

  projects: () =>
    request<ProjectListResponse>('/admin/projects?page=1&page_size=100', { auth: true }),
  project: (slug: string) =>
    request<ProjectDetail>(`/projects/${encodeURIComponent(slug)}`, { auth: true }),
  createProject: (body: ProjectCreateRequest) =>
    request<ProjectDetail>('/projects', { method: 'POST', auth: true, body }),
  updateProject: (slug: string, body: ProjectUpdateRequest) =>
    request<ProjectDetail>(`/projects/${encodeURIComponent(slug)}`, {
      method: 'PUT',
      auth: true,
      body,
    }),
  deleteProject: (slug: string) =>
    request<unknown>(`/projects/${encodeURIComponent(slug)}`, { method: 'DELETE', auth: true }),

  categories: () => request<Category[]>('/categories'),
  createCategory: (body: CategoryRequest) =>
    request<Category>('/categories', { method: 'POST', auth: true, body }),
  updateCategory: (slug: string, body: CategoryRequest) =>
    request<Category>(`/categories/${encodeURIComponent(slug)}`, {
      method: 'PUT',
      auth: true,
      body,
    }),
  deleteCategory: (slug: string) =>
    request<unknown>(`/categories/${encodeURIComponent(slug)}`, { method: 'DELETE', auth: true }),
  tags: () => request<Tag[]>('/tags'),
  createTag: (body: TagRequest) => request<Tag>('/tags', { method: 'POST', auth: true, body }),
  updateTag: (slug: string, body: TagRequest) =>
    request<Tag>(`/tags/${encodeURIComponent(slug)}`, { method: 'PUT', auth: true, body }),
  deleteTag: (slug: string) =>
    request<unknown>(`/tags/${encodeURIComponent(slug)}`, { method: 'DELETE', auth: true }),
```

`refreshPromise` 是并发控制，不是缓存。多个请求同时遇到 401 时只发一个刷新请求；刷新失败统一清理内存会话。

## 10-05 创建会话启动器和路由守卫

创建 `web/components/admin/auth-gate.tsx`，文件全文：

```tsx
'use client';

import { useEffect, useRef, type ReactNode } from 'react';
import { usePathname, useRouter } from 'next/navigation';
import { LoaderCircle } from 'lucide-react';
import { adminApi } from '@/lib/admin/api';
import { session, useSession } from '@/lib/admin/session';

export function AuthGate({ children }: { children: ReactNode }) {
  const current = useSession();
  const pathname = usePathname();
  const router = useRouter();
  const started = useRef(false);
  const loginPage = pathname === '/admin/login';

  useEffect(() => {
    if (started.current) return;
    started.current = true;
    void adminApi.restore().then(async (token) => {
      if (token === null) {
        session.clear();
        return;
      }
      try {
        session.setUser(await adminApi.me());
      } catch {
        session.clear();
      }
    });
  }, []);

  useEffect(() => {
    if (current.status === 'anonymous' && !loginPage) router.replace('/admin/login/');
    if (current.status === 'authenticated' && loginPage) router.replace('/admin/');
  }, [current.status, loginPage, router]);

  if (loginPage) return <>{children}</>;
  if (current.status !== 'authenticated') {
    return (
      <main className="admin-loading" aria-live="polite">
        <LoaderCircle className="spin" aria-hidden="true" />
        <span>正在验证会话</span>
      </main>
    );
  }
  return <>{children}</>;
}
```

## 10-07 创建登录页

创建 `web/app/admin/login/page.tsx`，文件全文：

```tsx
'use client';

import { useState, type FormEvent } from 'react';
import { useRouter } from 'next/navigation';
import { KeyRound, LoaderCircle } from 'lucide-react';
import { adminApi, ApiError } from '@/lib/admin/api';
import { session } from '@/lib/admin/session';

export default function LoginPage() {
  const router = useRouter();
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [submitting, setSubmitting] = useState(false);

  async function submit(event: FormEvent<HTMLFormElement>): Promise<void> {
    event.preventDefault();
    setError('');
    setSubmitting(true);
    try {
      const token = await adminApi.login(username, password);
      session.authenticate(token.access_token);
      session.setUser(await adminApi.me());
      router.replace('/admin/');
    } catch (cause) {
      setError(cause instanceof ApiError ? cause.message : '网络连接失败，请稍后重试');
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <main className="admin-loading">
      <form
        className="admin-panel admin-form"
        style={{ width: 'min(420px, calc(100vw - 28px))' }}
        onSubmit={(event) => void submit(event)}
      >
        <div>
          <h1>管理后台</h1>
          <p>使用管理员账号继续。</p>
        </div>
        {error && (
          <p className="notice error" role="alert">
            {error}
          </p>
        )}
        <div className="field">
          <label htmlFor="username">用户名</label>
          <input
            id="username"
            autoComplete="username"
            required
            maxLength={50}
            value={username}
            onChange={(event) => setUsername(event.target.value)}
          />
        </div>
        <div className="field">
          <label htmlFor="password">密码</label>
          <input
            id="password"
            type="password"
            autoComplete="current-password"
            required
            minLength={8}
            maxLength={100}
            value={password}
            onChange={(event) => setPassword(event.target.value)}
          />
        </div>
```

## 10-12 创建站点设置页

创建 `web/app/admin/settings/page.tsx`，文件全文：

```tsx
'use client';

import { useEffect, useState, type FormEvent } from 'react';
import { Save } from 'lucide-react';
import { adminApi, ApiError } from '@/lib/admin/api';
import type { SiteSettings } from '@/lib/admin/types';

const empty: SiteSettings = {
  site_title: '',
  site_description: '',
  site_keywords: '',
  footer_text: '',
  icp_number: null,
  about_content: '',
  social_links: [],
};

export default function SettingsPage() {
  const [form, setForm] = useState<SiteSettings>(empty);
  const [links, setLinks] = useState('[]');
  const [message, setMessage] = useState('');
  const [error, setError] = useState('');
  useEffect(() => {
    void adminApi
      .settings()
      .then((value) => {
        setForm(value);
        setLinks(JSON.stringify(value.social_links, null, 2));
      })
      .catch(() => setError('设置加载失败'));
  }, []);
  async function save(event: FormEvent) {
    event.preventDefault();
    setError('');
    setMessage('');
    try {
      const socialLinks = JSON.parse(links) as SiteSettings['social_links'];
      if (!Array.isArray(socialLinks)) throw new Error('社交链接必须是数组');
      const saved = await adminApi.updateSettings({ ...form, social_links: socialLinks });
      setForm(saved);
      setLinks(JSON.stringify(saved.social_links, null, 2));
      setMessage('站点设置已保存。重新发布静态站点后公共页面才会更新。');
    } catch (cause) {
      setError(cause instanceof ApiError || cause instanceof Error ? cause.message : '保存失败');
    }
  }
  return (
    <>
      <header className="admin-header">
        <div>
          <h1>站点设置</h1>
          <p>品牌、关于页和社交链接</p>
        </div>
      </header>
      {message && (
        <p className="notice" role="status">
          {message}
        </p>
      )}
      {error && (
        <p className="notice error" role="alert">
          {error}
        </p>
      )}
      <form className="admin-panel admin-form" onSubmit={(event) => void save(event)}>
        <div className="form-grid">
          <div className="field">
            <label htmlFor="site-title">站点标题</label>
            <input
              id="site-title"
              required
              value={form.site_title}
              onChange={(e) => setForm({ ...form, site_title: e.target.value })}
            />
          </div>
          <div className="field">
            <label htmlFor="site-keywords">关键词</label>
            <input
              id="site-keywords"
              value={form.site_keywords}
              onChange={(e) => setForm({ ...form, site_keywords: e.target.value })}
            />
          </div>
          <div className="field full">
            <label htmlFor="site-description">站点描述</label>
            <textarea
              id="site-description"
              value={form.site_description}
              onChange={(e) => setForm({ ...form, site_description: e.target.value })}
            />
          </div>
          <div className="field full">
            <label htmlFor="about-content">关于页 Markdown</label>
            <textarea
              className="editor"
              id="about-content"
              value={form.about_content}
              onChange={(e) => setForm({ ...form, about_content: e.target.value })}
            />
          </div>
          <div className="field">
            <label htmlFor="footer-text">页脚文字</label>
            <input
              id="footer-text"
              value={form.footer_text}
              onChange={(e) => setForm({ ...form, footer_text: e.target.value })}
            />
          </div>
          <div className="field">
            <label htmlFor="icp-number">ICP备案号</label>
            <input
              id="icp-number"
              value={form.icp_number ?? ''}
              onChange={(e) => setForm({ ...form, icp_number: e.target.value || null })}
            />
          </div>
          <div className="field full">
            <label htmlFor="social-links">社交链接 JSON</label>
            <textarea
              className="editor"
              id="social-links"
              spellCheck={false}
              value={links}
              onChange={(e) => setLinks(e.target.value)}
            />
          </div>
        </div>
        <button className="button" type="submit">
          <Save size={18} />
          保存设置
        </button>
      </form>
    </>
  );
}
```

社交链接 JSON 的每项必须是 `{"name":"GitHub","url":"https://github.com/你的账号","icon":"github"}` 形状。这里用 JSON 是为了练习结构化配置边界；后续可以在独立 PR 中替换为可增删的数组表单。
