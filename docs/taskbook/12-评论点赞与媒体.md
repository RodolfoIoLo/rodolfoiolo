import type { components } from '@/types/openapi';

export type CommentNode = components['schemas']['CommentNode'];
export type CommentListResponse = components['schemas']['CommentListResponse'];
export type CommentCreateRequest = components['schemas']['CommentCreateRequest'];
export type LikeStatus = components['schemas']['LikeStatus'];
type Envelope<T> = { code: number; message: string; data: T };

const baseURL = process.env.NEXT_PUBLIC_API_BASE_URL ?? 'http://localhost:8080/api/v1';

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const response = await fetch(`${baseURL}${path}`, {
    ...init,
    credentials: 'include',
    headers: {
      Accept: 'application/json',
      ...(init?.body ? { 'Content-Type': 'application/json' } : {}),
      ...(init?.headers ?? {}),
    },
  });
  const body = (await response.json().catch(() => null)) as Envelope<T> | null;
  if (!response.ok || body === null || body.code !== 0)
    throw new Error(body?.message ?? '请求失败');
  return body.data;
}

export const interactionApi = {
  likeStatus: (slug: string) =>
    request<LikeStatus>(`/articles/${encodeURIComponent(slug)}/like/status`),
  setLike: (slug: string, liked: boolean) =>
    request<LikeStatus>(`/articles/${encodeURIComponent(slug)}/like`, {
      method: 'PUT',
      headers: { 'X-Requested-With': 'XMLHttpRequest' },
      body: JSON.stringify({ action: liked ? 'unlike' : 'like' }),
    }),
  comments: (slug: string) =>
    request<CommentListResponse>(
      `/articles/${encodeURIComponent(slug)}/comments?page=1&page_size=100`,
    ),
  comment: (slug: string, body: CommentCreateRequest) =>
    request<unknown>(`/articles/${encodeURIComponent(slug)}/comments`, {
      method: 'POST',
      headers: { 'X-Requested-With': 'XMLHttpRequest' },
      body: JSON.stringify(body),
    }),
};
```

'use client';

import { adminApi } from '@/lib/admin/api';
import { session } from '@/lib/admin/session';
import type { components } from '@/types/openapi';

export type Comment = components['schemas']['Comment'];
type Envelope<T> = { code: number; message: string; data: T };
const baseURL = process.env.NEXT_PUBLIC_API_BASE_URL ?? 'http://localhost:8080/api/v1';

async function authenticated<T>(path: string, init: RequestInit = {}, retry = true): Promise<T> {
  const response = await fetch(`${baseURL}${path}`, {
    ...init,
    credentials: 'include',
    headers: {
      Accept: 'application/json',
      'X-Requested-With': 'XMLHttpRequest',
      ...(session.get().accessToken
        ? { Authorization: `Bearer ${session.get().accessToken}` }
        : {}),
      ...(init.headers ?? {}),
    },
  });
  if (response.status === 401 && retry && (await adminApi.restore()) !== null)
    return authenticated<T>(path, init, false);
  const body = (await response.json().catch(() => null)) as Envelope<T> | null;
  if (!response.ok || body === null || body.code !== 0)
    throw new Error(body?.message ?? '请求失败');
  return body.data;
}

export const moderationApi = {
  comments: (status = 'pending') =>
    authenticated<{ items: Comment[] }>(`/admin/comments?status=${status}&page=1&page_size=100`),
  approve: (id: number) =>
    authenticated<unknown>(`/admin/comments/${id}/approve`, { method: 'POST' }),
  reject: (id: number) =>
    authenticated<unknown>(`/admin/comments/${id}/reject`, { method: 'POST' }),
  remove: (id: number) => authenticated<unknown>(`/comments/${id}`, { method: 'DELETE' }),
  upload: (file: File) => {
    const data = new FormData();
    data.append('file', file);
    return authenticated<{
      url: string;
      filename: string;
      size: number;
      width: number;
      height: number;
    }>('/upload/image', { method: 'POST', body: data });
  },
};

package storage

import (
	"bytes"
	"context"
	"fmt"
	"strings"

	aliyun "github.com/aliyun/aliyun-oss-go-sdk/oss"
)

type OSS struct { bucket *aliyun.Bucket; publicBase string }

func NewOSS(endpoint, bucketName, accessKeyID, accessKeySecret, publicBase string) (*OSS, error) {
	client, err := aliyun.New(endpoint, accessKeyID, accessKeySecret)
	if err != nil { return nil, fmt.Errorf("create OSS client: %w", err) }
	bucket, err := client.Bucket(bucketName)
	if err != nil { return nil, fmt.Errorf("open OSS bucket: %w", err) }
	return &OSS{bucket: bucket, publicBase: strings.TrimRight(publicBase, "/")}, nil
}
func (s *OSS) Put(ctx context.Context, key, contentType string, data []byte) error {
	if err := s.bucket.PutObject(key, bytes.NewReader(data), aliyun.ContentType(contentType), aliyun.WithContext(ctx)); err != nil { return fmt.Errorf("put OSS object: %w", err) }
	return nil
}
func (s *OSS) Delete(ctx context.Context, key string) error {
	if err := s.bucket.DeleteObject(key, aliyun.WithContext(ctx)); err != nil { return fmt.Errorf("delete OSS object: %w", err) }
	return nil
}
func (s *OSS) PublicURL(key string) string { return s.publicBase + "/" + strings.TrimLeft(key, "/") }
```

package storage

import (
	"fmt"
	"os"
	"strings"
)

func NewFromEnvironment(localRoot, localPublicURL string) (ObjectStore, error) {
	driver := strings.ToLower(strings.TrimSpace(os.Getenv("STORAGE_DRIVER")))
	if driver == "" || driver == "local" { return NewLocal(localRoot, localPublicURL) }
	if driver != "oss" { return nil, fmt.Errorf("unsupported STORAGE_DRIVER %q", driver) }
	values := map[string]string{
		"OSS_ENDPOINT": os.Getenv("OSS_ENDPOINT"), "OSS_BUCKET": os.Getenv("OSS_BUCKET"),
		"OSS_ACCESS_KEY_ID": os.Getenv("OSS_ACCESS_KEY_ID"), "OSS_ACCESS_KEY_SECRET": os.Getenv("OSS_ACCESS_KEY_SECRET"),
		"OSS_PUBLIC_URL": os.Getenv("OSS_PUBLIC_URL"),
	}
	for name, value := range values { if strings.TrimSpace(value) == "" { return nil, fmt.Errorf("%s is required for OSS storage", name) } }
	return NewOSS(values["OSS_ENDPOINT"], values["OSS_BUCKET"], values["OSS_ACCESS_KEY_ID"], values["OSS_ACCESS_KEY_SECRET"], values["OSS_PUBLIC_URL"])
}

在 ECS 的 `.env.production` 增加 `STORAGE_DRIVER=oss`、`OSS_ENDPOINT`、`OSS_BUCKET`、`OSS_ACCESS_KEY_ID`、`OSS_ACCESS_KEY_SECRET`、`OSS_PUBLIC_URL` 并 `chmod 600`，Compose 的 API environment 显式映射这六项。Bucket 保持私有；`OSS_PUBLIC_URL` 必须是启用了私有 Bucket 回源鉴权的 CDN 自定义 HTTPS 域名，不是 OSS 匿名直链。重建 API 后上传一张测试图，核对数据库 hash/尺寸、OSS 对象 Content-Type、图片无 EXIF、CDN URL 可读、直接 OSS 匿名 URL 返回拒绝。
